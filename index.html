    <script>

        let aholilar = [];

        let pastStavka = 0;

        let balandStavka = 0;

        let oylar = []; // Oylik ma'lumotlar massivi


        // LocalStorage dan yuklash

        if (localStorage.getItem('aholilar')) {

            aholilar = JSON.parse(localStorage.getItem('aholilar'));

        }

        if (localStorage.getItem('pastStavka')) {

            pastStavka = parseInt(localStorage.getItem('pastStavka'));

            document.getElementById('pastStavka').value = pastStavka;

        }

        if (localStorage.getItem('balandStavka')) {

            balandStavka = parseInt(localStorage.getItem('balandStavka'));

            document.getElementById('balandStavka').value = balandStavka;

        }

        if (localStorage.getItem('oylar')) {

            oylar = JSON.parse(localStorage.getItem('oylar'));

        }


        // Jadvalni yangilash funksiyasi

        function jadvalniYangila() {

            const tbody = document.querySelector('#aholiTable tbody');

            tbody.innerHTML = '';

            aholilar.forEach(aholi => {

                const umumiy = aholi.molMulki + aholi.daromad + aholi.jamgarma; // Soddalashtirilgan umumiy qiymat (mol, yer va h.k. qiymatini qo'shish mumkin)

                const tr = document.createElement('tr');

                tr.innerHTML = `

                    <td>${aholi.ism}</td>

                    <td>${aholi.molMulki}</td>

                    <td>${aholi.mol}</td>

                    <td>${aholi.jonzotlar}</td>

                    <td>${aholi.daromad}</td>

                    <td>${aholi.uyHolati}</td>

                    <td>${aholi.yerO'lchami}</td>

                    <td>${aholi.jamgarma}</td>

                    <td>${umumiy}</td>

                `;

                if (umumiy < pastStavka) {

                    tr.classList.add('qizil');

                } else if (umumiy > balandStavka) {

                    tr.classList.add('yashil');

                } else {

                    tr.classList.add('sariq');

                }

                tbody.appendChild(tr);

            });

        }


        // Tendensiyani hisoblash

        function tendensiyaniYangila() {

            if (oylar.length < 2) {

                document.getElementById('tendensiyaMatn').innerText = 'Tendensiya uchun kamida 2 oy ma\'lumoti kerak.';

                return;

            }

            const oxirgiOy = oylar[oylar.length - 1];

            const oldingiOy = oylar[oylar.length - 2];


            const yashilOzg = oxirgiOy.yashil - oldingiOy.yashil;

            const sariqOzg = oxirgiOy.sariq - oldingiOy.sariq;

            const qizilOzg = oxirgiOy.qizil - oldingiOy.qizil;


            let matn = `Yashil kategoriya: ${yashilOzg > 0 ? 'ko\'tarilgan (+' + yashilOzg + ')' : 'tushgan (' + yashilOzg + ')'}\n`;

            matn += `Sariq kategoriya: ${sariqOzg > 0 ? 'ko\'tarilgan (+' + sariqOzg + ')' : 'tushgan (' + sariqOzg + ')'}\n`;

            matn += `Qizil kategoriya: ${qizilOzg > 0 ? 'ko\'tarilgan (+' + qizilOzg + ')' : 'tushgan (' + qizilOzg + ')'}`;


            document.getElementById('tendensiyaMatn').innerText = matn;

        }


        // Aholi qo'shish

        document.getElementById('aholiForm').addEventListener('submit', function(e) {

            e.preventDefault();

            const aholi = {

                ism: document.getElementById('ism').value,

                molMulki: parseInt(document.getElementById('molMulki').value),

                mol: parseInt(document.getElementById('mol').value),

                jonzotlar: parseInt(document.getElementById('jonzotlar').value),

                daromad: parseInt(document.getElementById('daromad').value),

                uyHolati: document.getElementById('uyHolati').value,

                yerO'lchami: parseFloat(document.getElementById('yerO\'lchami').value),

                jamgarma: parseInt(document.getElementById('jamgarma').value)

            };

            aholilar.push(aholi);

            localStorage.setItem('aholilar', JSON.stringify(aholilar));

            jadvalniYangila();

            this.reset();

        });


        // Stavkalar belgilash

        document.getElementById('stavkaForm').addEventListener('submit', function(e) {

            e.preventDefault();

            pastStavka = parseInt(document.getElementById('pastStavka').value);

            balandStavka = parseInt(document.getElementById('balandStavka').value);

            localStorage.setItem('pastStavka', pastStavka);

            localStorage.setItem('balandStavka', balandStavka);

            jadvalniYangila();

        });


        // Joriy oyni saqlash

        document.getElementById('oyniSaqlash').addEventListener('click', function() {

            let yashil = 0, sariq = 0, qizil = 0;

            aholilar.forEach(aholi => {

                const umumiy = aholi.molMulki + aholi.daromad + aholi.jamgarma;

                if (umumiy < pastStavka) qizil++;

                else if (umumiy > balandStavka) yashil++;

                else sariq++;

            });

            oylar.push({ yashil, sariq, qizil });

            localStorage.setItem('oylar', JSON.stringify(oylar));

            tendensiyaniYangila();

        });


        // Dastlabki yuklash

        jadvalniYangila();

        tendensiyaniYangila();

    </script>
